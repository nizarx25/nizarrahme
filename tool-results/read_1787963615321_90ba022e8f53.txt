   115→function useFeaturedDomains() {
   116→  return useQuery<PublicDomain[]>({
   117→    queryKey: ['featured'],
   118→    queryFn: async () => {
   119→      const res = await fetch('/api/domains/featured')
   120→      if (!res.ok) throw new Error('Failed to fetch featured domains')
   121→      const data = await res.json()
   122→      return data.domains as PublicDomain[]
   123→    },
   124→  })
   125→}
   126→
   127→function useDomains(
   128→  search: string,
   129→  category: string,
   130→  extension: string,
   131→  status: string,
   132→  featured: boolean,
   133→  hasPrice: boolean,
   134→  sort: string,
   135→  page: number
   136→) {
   137→  const params = new URLSearchParams()
   138→  if (search) params.set('search', search)
   139→  if (category) params.set('category', category)
   140→  if (extension) params.set('extension', extension)
   141→  if (status) params.set('status', status)
   142→  if (featured) params.set('featured', 'true')
   143→  if (hasPrice) params.set('hasPrice', 'true')
   144→  if (sort) params.set('sort', sort)
   145→  params.set('page', String(page))
   146→  params.set('limit', '12')
   147→
   148→  return useQuery<{
   149→    domains: PublicDomain[]
   150→    total: number
   151→    page: number
   152→    limit: number
   153→    categories: string[]
   154→    extensions: string[]
   155→  }>({
   156→    queryKey: ['domains', search, category, extension, status, featured, hasPrice, sort, page],
   157→    queryFn: async () => {
   158→      const res = await fetch(`/api/domains?${params.toString()}`)
   159→      if (!res.ok) throw new Error('Failed to fetch domains')
   160→      return res.json()
   161→    },
   162→  })
   163→}
   164→
   165→function useDomainDetail(slug: string | null) {
   166→  return useQuery<{ domain: PublicDomain; relatedDomains: PublicDomain[] } | null>({
   167→    queryKey: ['domain', slug],
   168→    queryFn: async () => {
   169→      if (!slug) return null
   170→      const res = await fetch(`/api/domains/${slug}`)
   171→      if (!res.ok) throw new Error('Failed to fetch domain')
   172→      return res.json()
   173→    },
   174→    enabled: !!slug,
   175→  })
   176→}

   211→function useSubmitInquiry() {
   212→  const queryClient = useQueryClient()
   213→  const { toast } = useToast()
   214→
   215→  return useMutation({
   216→    mutationFn: async (data: Record<string, unknown>) => {
   217→      const res = await fetch('/api/inquiries', {
   218→        method: 'POST',
   219→        headers: { 'Content-Type': 'application/json' },
   220→        body: JSON.stringify(data),
   221→      })
   222→      const json = await res.json()
   223→      if (!res.ok) {
   224→        if (json.details) throw new Error(JSON.stringify(json.details))
   225→        throw new Error(json.error || 'Submission failed')
   226→      }
   227→      return json
   228→    },
   229→    onSuccess: () => {
   230→      toast({ title: 'Inquiry sent', description: 'Thank you. Your inquiry has been received.' })
   231→      queryClient.invalidateQueries({ queryKey: ['domains'] })
   232→    },
   233→    onError: (err) => {
   234→      toast({ title: 'Submission failed', description: err.message, variant: 'destructive' })
   235→    },
   236→  })
   237→}
