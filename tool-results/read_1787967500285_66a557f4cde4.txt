   129→function useFeaturedDomains() {
   130→  return useQuery<PublicDomain[]>({
   131→    queryKey: ['featured'],
   132→    queryFn: async () => {
   133→      const res = await fetch('/api/domains/featured')
   134→      if (!res.ok) throw new Error('Failed to fetch featured domains')
   135→      const data = await res.json()
   136→      return data.domains as PublicDomain[]
   137→    },
   138→  })
   139→}

   141→function useDomains(
   142→  search: string,
   143→  category: string,
   144→  extension: string,
   145→  status: string,
   146→  featured: boolean,
   147→  hasPrice: boolean,
   148→  sort: string,
   149→  page: number
   150→) {
   151→  const params = new URLSearchParams()
   152→  if (search) params.set('search', search)
   153→  if (category) params.set('category', category)
   154→  if (extension) params.set('extension', extension)
   155→  if (status) params.set('status', status)
   156→  if (featured) params.set('featured', 'true')
   157→  if (hasPrice) params.set('hasPrice', 'true')
   158→  if (sort) params.set('sort', sort)
   159→  params.set('page', String(page))
   160→  params.set('limit', '12')
   161→
   162→  return useQuery<{
   163→    domains: PublicDomain[]
   164→    total: number
   165→    page: number
   166→    limit: number
   167→    categories: string[]
   168→    extensions: string[]
   169→  }>({
   170→    queryKey: ['domains', search, category, extension, status, featured, hasPrice, sort, page],
   171→    queryFn: async () => {
   172→      const res = await fetch(`/api/domains?${params.toString()}`)
   173→      if (!res.ok) throw new Error('Failed to fetch domains')
   174→      return res.json()
   175→    },
   176→  })
   177→}

   225→function useSubmitInquiry() {
   226→  const queryClient = useQueryClient()
   227→  const { toast } = useToast()
   228→
   229→  return useMutation({
   230→    mutationFn: async (data: Record<string, unknown>) => {
   231→      const res = await fetch('/api/inquiries', {
   232→        method: 'POST',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify(data),
   235→      })
   236→      const json = await res.json()
   237→      if (!res.ok) {
   238→        if (json.details) throw new Error(JSON.stringify(json.details))
   239→        throw new Error(json.error || 'Submission failed')
   240→      }
   241→      return json
   242→    },
   243→    onSuccess: () => {
   244→      toast({ title: 'Inquiry sent', description: 'Thank you. Your inquiry has been received.' })
   245→      queryClient.invalidateQueries({ queryKey: ['domains'] })
   246→    },
   247→    onError: (err) => {
   248→      toast({ title: 'Submission failed', description: err.message, variant: 'destructive' })
   249→    },
   250→  })
   251→}
