   117→function useFeaturedDomains() {
   118→  return useQuery<PublicDomain[]>({
   119→    queryKey: ['featured'],
   120→    queryFn: async () => {
   121→      const res = await fetch('/api/domains/featured')
   122→      if (!res.ok) throw new Error('Failed to fetch featured domains')
   123→      const data = await res.json()
   124→      return data.domains as PublicDomain[]
   125→    },
   126→  })
   127→}
   128→
   129→function useDomains(
   130→  search: string,
   131→  category: string,
   132→  extension: string,
   133→  status: string,
   134→  featured: boolean,
   135→  hasPrice: boolean,
   136→  sort: string,
   137→  page: number
   138→) {
   139→  const params = new URLSearchParams()
   140→  if (search) params.set('search', search)
   141→  if (category) params.set('category', category)
   142→  if (extension) params.set('extension', extension)
   143→  if (status) params.set('status', status)
   144→  if (featured) params.set('featured', 'true')
   145→  if (hasPrice) params.set('hasPrice', 'true')
   146→  if (sort) params.set('sort', sort)
   147→  params.set('page', String(page))
   148→  params.set('limit', '12')
   149→
   150→  return useQuery<{
   151→    domains: PublicDomain[]
   152→    total: number
   153→    page: number
   154→    limit: number
   155→    categories: string[]
   156→    extensions: string[]
   157→  }>({
   158→    queryKey: ['domains', search, category, extension, status, featured, hasPrice, sort, page],
   159→    queryFn: async () => {
   160→      const res = await fetch(`/api/domains?${params.toString()}`)
   161→      if (!res.ok) throw new Error('Failed to fetch domains')
   162→      return res.json()
   163→    },
   164→  })
   165→}
   166→
   167→function useDomainDetail(slug: string | null) {
   168→  return useQuery<{ domain: PublicDomain; relatedDomains: PublicDomain[] } | null>({
   169→    queryKey: ['domain', slug],
   170→    queryFn: async () => {
   171→      if (!slug) return null
   172→      const res = await fetch(`/api/domains/${slug}`)
   173→      if (!res.ok) throw new Error('Failed to fetch domain')
   174→      return res.json()
   175→    },
   176→    enabled: !!slug,
   177→  })
   178→}

   213→function useSubmitInquiry() {
   214→  const queryClient = useQueryClient()
   215→  const { toast } = useToast()
   216→
   217→  return useMutation({
   218→    mutationFn: async (data: Record<string, unknown>) => {
   219→      const res = await fetch('/api/inquiries', {
   220→        method: 'POST',
   221→        headers: { 'Content-Type': 'application/json' },
   222→        body: JSON.stringify(data),
   223→      })
   224→      const json = await res.json()
   225→      if (!res.ok) {
   226→        if (json.details) throw new Error(JSON.stringify(json.details))
   227→        throw new Error(json.error || 'Submission failed')
   228→      }
   229→      return json
   230→    },
   231→    onSuccess: () => {
   232→      toast({ title: 'Inquiry sent', description: 'Thank you. Your inquiry has been received.' })
   233→      queryClient.invalidateQueries({ queryKey: ['domains'] })
   234→    },
   235→    onError: (err) => {
   236→      toast({ title: 'Submission failed', description: err.message, variant: 'destructive' })
   237→    },
   238→  })
   239→}

   469→function ClaimedTicker() {
   470→  const { data: transactions } = useTransactions()
   471→  const tickerRef = useRef<HTMLDivElement>(null)
   472→
   473→  if (!transactions || transactions.transactions.length === 0) return null
   474→
   475→  const items = transactions.transactions.map((tx) => ({
   476→    domain: tx.domain,
   477→    label: tx.status === 'Sold' ? 'SOLD' : 'CLAIMED',
   478→  }))
   479→
   480→  // Double for seamless loop
   481→  const doubled = [...items, ...items]
   482→
   483→  return (
   484→    <div className="relative overflow-hidden py-4 bg-[#0B211E]/50 surface-border-y" aria-label="Recent domain activity">
   485→      <div ref={tickerRef} className="animate-ticker flex gap-8 whitespace-nowrap w-max">
   486→        {doubled.map((item, i) => (
   487→          <div key={i} className="flex items-center gap-3 shrink-0">
   488→            <CircleDot className="size-3 text-teal" />
   489→            <span className="font-display text-sm font-bold text-[#B8C8C4]">{item.domain}</span>
   490→            <span className="text-xs font-mono-accent text-teal tracking-wider">{item.label}</span>
   491→          </div>
   492→        ))}
   493→      </div>
   494→    </div>
   495→  )
   496→}
