   128→function useFeaturedDomains() {
   129→  return useQuery<PublicDomain[]>({
   130→    queryKey: ['featured'],
   131→    queryFn: async () => {
   132→      const res = await fetch('/api/domains/featured')
   133→      if (!res.ok) throw new Error('Failed to fetch featured domains')
   134→      const data = await res.json()
   135→      return data.domains as PublicDomain[]
   136→    },
   137→  })
   138→}
   139→
   140→function useDomains(
   141→  search: string,
   142→  category: string,
   143→  extension: string,
   144→  status: string,
   145→  featured: boolean,
   146→  hasPrice: boolean,
   147→  sort: string,
   148→  page: number
   149→) {
   150→  const params = new URLSearchParams()
   151→  if (search) params.set('search', search)
   152→  if (category) params.set('category', category)
   153→  if (extension) params.set('extension', extension)
   154→  if (status) params.set('status', status)
   155→  if (featured) params.set('featured', 'true')
   156→  if (hasPrice) params.set('hasPrice', 'true')
   157→  if (sort) params.set('sort', sort)
   158→  params.set('page', String(page))
   159→  params.set('limit', '12')
   160→
   161→  return useQuery<{
   162→    domains: PublicDomain[]
   163→    total: number
   164→    page: number
   165→    limit: number
   166→    categories: string[]
   167→    extensions: string[]
   168→  }>({
   169→    queryKey: ['domains', search, category, extension, status, featured, hasPrice, sort, page],
   170→    queryFn: async () => {
   171→      const res = await fetch(`/api/domains?${params.toString()}`)
   172→      if (!res.ok) throw new Error('Failed to fetch domains')
   173→      return res.json()
   174→    },
   175→  })
   176→}

   224→function useSubmitInquiry() {
   225→  const queryClient = useQueryClient()
   226→  const { toast } = useToast()
   227→
   228→  return useMutation({
   229→    mutationFn: async (data: Record<string, unknown>) => {
   230→      const res = await fetch('/api/inquiries', {
   231→        method: 'POST',
   232→        headers: { 'Content-Type': 'application/json' },
   233→        body: JSON.stringify(data),
   234→      })
   235→      const json = await res.json()
   236→      if (!res.ok) {
   237→        if (json.details) throw new Error(JSON.stringify(json.details))
   238→        throw new Error(json.error || 'Submission failed')
   239→      }
   240→      return json
   241→    },
   242→    onSuccess: () => {
   243→      toast({ title: 'Inquiry sent', description: 'Thank you. Your inquiry has been received.' })
   244→      queryClient.invalidateQueries({ queryKey: ['domains'] })
   245→    },
   246→    onError: (err) => {
   247→      toast({ title: 'Submission failed', description: err.message, variant: 'destructive' })
   248→    },
   249→  })
   250→}

   582→function FeaturedCarousel({ domains }: { domains: PublicDomain[] }) {
   583→  const [current, setCurrent] = useState(0)
   584→  const nav = useNavigation()
   585→  const intervalRef = useRef<NodeJS.Timeout | null>(null)
   586→
   587→  const startAutoplay = useCallback(() => {
   588→    if (intervalRef.current) clearInterval(intervalRef.current)
   589→    intervalRef.current = setInterval(() => {
   590→      setCurrent((prev) => (prev + 1) % domains.length)
   591→    }, 4000)
   592→  }, [domains.length])
   593→
   594→  useEffect(() => {
   595→    startAutoplay()
   596→    return () => { if (intervalRef.current) clearInterval(intervalRef.current) }
   597→  }, [startAutoplay])
