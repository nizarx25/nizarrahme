   127→function useFeaturedDomains() {
   128→  return useQuery<PublicDomain[]>({
   129→    queryKey: ['featured'],
   130→    queryFn: async () => {
   131→      const res = await fetch('/api/domains/featured')
   132→      if (!res.ok) throw new Error('Failed to fetch featured domains')
   133→      const data = await res.json()
   134→      return data.domains as PublicDomain[]
   135→    },
   136→  })
   137→}
   138→
   139→function useDomains(
   140→  search: string,
   141→  category: string,
   142→  extension: string,
   143→  status: string,
   144→  featured: boolean,
   145→  hasPrice: boolean,
   146→  sort: string,
   147→  page: number
   148→) {
   149→  const params = new URLSearchParams()
   150→  if (search) params.set('search', search)
   151→  if (category) params.set('category', category)
   152→  if (extension) params.set('extension', extension)
   153→  if (status) params.set('status', status)
   154→  if (featured) params.set('featured', 'true')
   155→  if (hasPrice) params.set('hasPrice', 'true')
   156→  if (sort) params.set('sort', sort)
   157→  params.set('page', String(page))
   158→  params.set('limit', '12')
   159→
   160→  return useQuery<{
   161→    domains: PublicDomain[]
   162→    total: number
   163→    page: number
   164→    limit: number
   165→    categories: string[]
   166→    extensions: string[]
   167→  }>({
   168→    queryKey: ['domains', search, category, extension, status, featured, hasPrice, sort, page],
   169→    queryFn: async () => {
   170→      const res = await fetch(`/api/domains?${params.toString()}`)
   171→      if (!res.ok) throw new Error('Failed to fetch domains')
   172→      return res.json()
   173→    },
   174→  })
   175→}
   176→
   177→function useDomainDetail(slug: string | null) {
   178→  return useQuery<{ domain: PublicDomain; relatedDomains: PublicDomain[] } | null>({
   179→    queryKey: ['domain', slug],
   180→    queryFn: async () => {
   181→      if (!slug) return null
   182→      const res = await fetch(`/api/domains/${slug}`)
   183→      if (!res.ok) throw new Error('Failed to fetch domain')
   184→      return res.json()
   185→    },
   186→    enabled: !!slug,
   187→  })
   188→}

   223→function useSubmitInquiry() {
   224→  const queryClient = useQueryClient()
   225→  const { toast } = useToast()
   226→
   227→  return useMutation({
   228→    mutationFn: async (data: Record<string, unknown>) => {
   229→      const res = await fetch('/api/inquiries', {
   230→        method: 'POST',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify(data),
   233→      })
   234→      const json = await res.json()
   235→      if (!res.ok) {
   236→        if (json.details) throw new Error(JSON.stringify(json.details))
   237→        throw new Error(json.error || 'Submission failed')
   238→      }
   239→      return json
   240→    },
   241→    onSuccess: () => {
   242→      toast({ title: 'Inquiry sent', description: 'Thank you. Your inquiry has been received.' })
   243→      queryClient.invalidateQueries({ queryKey: ['domains'] })
   244→    },
   245→    onError: (err) => {
   246→      toast({ title: 'Submission failed', description: err.message, variant: 'destructive' })
   247→    },
   248→  })
   249→}
