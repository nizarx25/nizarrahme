   136→function useFeaturedDomains() {
   137→  return useQuery<PublicDomain[]>({
   138→    queryKey: ['featured'],
   139→    queryFn: async () => {
   140→      const res = await fetch('/api/domains/featured')
   141→      if (!res.ok) throw new Error('Failed to fetch featured domains')
   142→      const data = await res.json()
   143→      return data.domains as PublicDomain[]
   144→    },
   145→  })
   146→}

   148→function useDomains(
   149→  search: string,
   150→  category: string,
   151→  extension: string,
   152→  status: string,
   153→  featured: boolean,
   154→  hasPrice: boolean,
   155→  sort: string,
   156→  page: number
   157→) {
   158→  const params = new URLSearchParams()
   159→  if (search) params.set('search', search)
   160→  if (category) params.set('category', category)
   161→  if (extension) params.set('extension', extension)
   162→  if (status) params.set('status', status)
   163→  if (featured) params.set('featured', 'true')
   164→  if (hasPrice) params.set('hasPrice', 'true')
   165→  if (sort) params.set('sort', sort)
   166→  params.set('page', String(page))
   167→  params.set('limit', '12')
   168→
   169→  return useQuery<{
   170→    domains: PublicDomain[]
   171→    total: number
   172→    page: number
   173→    limit: number
   174→    categories: string[]
   175→    extensions: string[]
   176→  }>({
   177→    queryKey: ['domains', search, category, extension, status, featured, hasPrice, sort, page],
   178→    queryFn: async () => {
   179→      const res = await fetch(`/api/domains?${params.toString()}`)
   180→      if (!res.ok) throw new Error('Failed to fetch domains')
   181→      return res.json()
   182→    },
   183→  })
   184→}

   192→      if (!res.ok) throw new Error('Failed to fetch domain')
   193→      return res.json()
   194→    },
   195→    enabled: !!slug,
   196→  })
   197→}

   232→function useSubmitInquiry() {
   233→  const queryClient = useQueryClient()
   234→  const { toast } = useToast()
   235→
   236→  return useMutation({
   237→    mutationFn: async (data: Record<string, unknown>) => {
   238→      const res = await fetch('/api/inquiries', {
   239→        method: 'POST',
   240→        headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify(data),
   242→      })
   243→      const json = await res.json()
   244→      if (!res.ok) {
   245→        if (json.details) throw new Error(JSON.stringify(json.details))
   246→        throw new Error(json.error || 'Submission failed')
   247→      }
   248→      return json
   249→    },
   250→    onSuccess: () => {
   251→      toast({ title: 'Inquiry sent', description: 'Thank you. Your inquiry has been received.' })
   252→      queryClient.invalidateQueries({ queryKey: ['domains'] })
   253→    },
   254→    onError: (err) => {
   255→      toast({ title: 'Submission failed', description: err.message, variant: 'destructive' })
   256→    },
   257→  })
   258→}
